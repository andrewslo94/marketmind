{# Reusable macro: renders one form field with its label and any validation errors. Called from register.html and login.html. #} {% macro render_field(field, autocomplete=None) %}
{{ field.label(class="form-label") }} {{ field(class="form-control" + (" is-invalid" if field.errors else ""), autocomplete=autocomplete) }} {% for error in field.errors %}
{{ error }}
{% endfor %}
{% endmacro %}
